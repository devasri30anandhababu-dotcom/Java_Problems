   import java.util.*;
public class arr3{
    public static void main(String[]args){
        Scanner sc=new Scanner(System.in);
        int n=sc.nextInt();
        double [] price=new double[n];
        for(int i=0;i<n;i++){
            price[i]=sc.nextDouble();
        }
        double max=0;
        double min=price[0];
        for(int j=0;j<n;j++){
            if(price[j]>max){
                max=price[j];
            }
        }
        for(int k=0;k<n;k++){
            if(price[k]<min){
                min=price[k];
            }
        }
        double diff=max-min;
        double per=diff/max*100.0;
        System.out.println("Number if Sellers: "+n);
        System.out.println("Lowest Price: $ "+min);
        System.out.println("Highest Price: $ "+max);
        System.out.println("Price Difference: $"+String.format("%.2f",diff));
        System.out.println("Savings: "+String.format("%.2f",per)+"%");
        }
    }
