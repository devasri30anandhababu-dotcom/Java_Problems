import java.util.*;
public class ar1{
    public static void main(String[]args){
        Scanner sc=new Scanner(System.in);
        int r=sc.nextInt();
        int c=sc.nextInt();
        int [][] mark=new int[r][c];
        for(int i=0;i<r;i++){
            for(int j=0;j<c;j++){
                mark[i][j]=sc.nextInt();
            }
        }
        System.out.println("Student Averages:");
        for(int i=0;i<r;i++){
            int total=0;
            for(int j=0;j<c;j++){
                total=total+mark[i][j];
            }
            double avg=(double)total/c;
            int finalavg=(int)(avg*100);
            System.out.println("Student"+(i+1)+": "+finalavg);
        }
          System.out.println("Subject Averages:");

            for(int j=0;j<c;j++){
                int total1=0;
                for(int i=0;i<r;i++){
                    total1=total1+mark[i][j];
                }
                double avg1=(double)total1/r;
                int finalavg1=(int)(avg1*100);
                System.out.println("Subject"+(j+1)+": "+finalavg1);
            }
        }
    }
