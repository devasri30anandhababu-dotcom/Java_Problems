import java.util.*;

public class array9 {
    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        int r = sc.nextInt();
        int c = sc.nextInt();

        int[][] sale = new int[r][c];

        for(int i = 0; i < r; i++) {
            for(int j = 0; j < c; j++) {
                sale[i][j] = sc.nextInt();
            }
        }
        System.out.println("Sales Matrix:");

        for(int i = 0; i < r; i++) {
            for(int j = 0; j < c; j++) {
                System.out.print(sale[i][j] + " ");
            }
            System.out.println();
        }
        System.out.println("Representative Totals:");

        for(int i = 0; i < r; i++) {

            int cadd = 0;

            for(int j = 0; j < c; j++) {
                cadd = cadd + sale[i][j];
            }

            System.out.println("Rep " + i + " : " + cadd);
        }

        System.out.println("Territory Totals:");

        for(int j = 0; j < c; j++) {

            int radd = 0;

            for(int i = 0; i < r; i++) {
                radd = radd + sale[i][j];
            }

            System.out.println("Territory " + j + " : " + radd);
        }
    }
}
