import java.util.*;
public class while13 {
    public static void main(String[] args) {
        Scanner sc=new Scanner(System.in);
        int base=sc.nextInt();
        int exponent=sc.nextInt();
        int res=1;
        int temp=exponent;
        while(temp>0){
            res=res*base;
            temp--;
        }
        System.out.println(base+"^"+exponent+"="+res);
    
        
    }
}
