
import java.util.*;
public class while5 {
    public static void main(String[]args){
        Scanner sc=new Scanner(System.in);
        int num=sc.nextInt();
        int rem=0;
        int rev=0;
        int original=num;
        while(num>0){
            rem=num%10;
            rev=(rev*10)+rem;
            
            num=num/10;
            
        }
        System.out.println("Original:"+ original);
        System.out.println("Reversed:"+rev);
    }
}
