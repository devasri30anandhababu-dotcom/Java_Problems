import java.util.*;

public class nested13 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int std = sc.nextInt();
        int sub = sc.nextInt();   
        sc.nextLine();

        System.out.println("Grade Sheet:");
        System.out.println("Student Sub1 Sub2 Sub3 Average");

        for (int i = 1; i <= std; i++) {
            String name = sc.nextLine();  
            int total = 0;

            System.out.print(name + " ");

            for (int j = 1; j <= sub; j++) {
                int m = sc.nextInt();   // marks for each subject]\
                total += m;
                System.out.print(m + " ");
            }
            sc.nextLine(); // clear buffer

            int avg = total / sub;
            System.out.println(avg);
        }
    }
}
